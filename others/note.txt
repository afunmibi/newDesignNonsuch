database for enrolment form
CREATE DATABASE IF NOT EXISTS nonsuch_portal;


CREATE TABLE IF NOT EXISTS enrolment (
    id INT AUTO_INCREMENT PRIMARY KEY,
    organization VARCHAR(100) NOT NULL,
    sname VARCHAR(50) NOT NULL,
    oname VARCHAR(50) NOT NULL,
    policy_no VARCHAR(50) NOT NULL,
    phone_no VARCHAR(20) NOT NULL,
    email VARCHAR(100) NOT NULL,
    provider VARCHAR(50) NOT NULL,
    gender ENUM('Male', 'Female') NOT NULL,
    plan_type ENUM('Gold', 'Silver', 'Platinum') NOT NULL,
    location VARCHAR(100) NOT NULL,
    dob DATE NOT NULL,
    no_of_dependant INT NOT NULL,
    reg_status ENUM('active', 'inactive') NOT NULL,
    alternate_no VARCHAR(20),
    date_captured DATETIME NOT NULL,
    photo VARCHAR(255) NOT NULL
);


-- Table: log_enrollees_in
CREATE TABLE IF NOT EXISTS log_enrollees_in (
    pa_code VARCHAR(50) PRIMARY KEY,
    name_of_enrollee VARCHAR(255) NOT NULL,
    nhia_no VARCHAR(100),
    phone VARCHAR(20),
    sex VARCHAR(10),
    dob DATE,
    no_of_days_admission INT,
    primary_hospital VARCHAR(255),
    secondary_hospital VARCHAR(255),
    primary_hospital_code VARCHAR(50),
    secondary_hospital_code VARCHAR(50),
    diagnosis TEXT,
    procedure_performed TEXT,
    further_diagnosis TEXT,
    -- Add other relevant columns from your form as needed
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Table: e_service_entry
CREATE TABLE IF NOT EXISTS e_service_entry (
    service_id INT AUTO_INCREMENT PRIMARY KEY,
    pa_code VARCHAR(50) NOT NULL,
    name_of_service VARCHAR(255),
    nhia_tariff DECIMAL(10, 2),
    hcf_amount_claimed DECIMAL(10, 2),
    amount_due DECIMAL(10, 2),
    qty INT,
    ten_percent_deduction DECIMAL(10, 2),
    total DECIMAL(10, 2),
    remarks VARCHAR(255),
    -- Add other relevant columns for service entries
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (pa_code) REFERENCES log_enrollees_in(pa_code) ON DELETE CASCADE
);

-- Table: e_drug_entry
CREATE TABLE IF NOT EXISTS e_drug_entry (
    drug_id INT AUTO_INCREMENT PRIMARY KEY,
    pa_code VARCHAR(50) NOT NULL,
    name_of_drug VARCHAR(255),
    nhia_tariff DECIMAL(10, 2),
    hcf_amount_claimed DECIMAL(10, 2),
    amount_due DECIMAL(10, 2),
    qty INT,
    ten_percent_deduction DECIMAL(10, 2),
    total DECIMAL(10, 2),
    remarks VARCHAR(255),
    -- Add other relevant columns for drug entries
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (pa_code) REFERENCES log_enrollees_in(pa_code) ON DELETE CASCADE
);
